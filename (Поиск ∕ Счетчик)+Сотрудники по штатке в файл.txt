//Конфигурация
var skipEngineers = '+'; //'+' - исключить из выгрузки сотрудников с почтой *example1*, *example2*, *example3* и с *info* в имени. '-' - не исключать
var isBoss = '-'; //'+' - записывать сотрудников только с руководящими должностями, '-' - записывать всех
var printRoleProf = '-'; //'+' - записывать профили сотрудников, '-' - не записывать
var noEmpty = '-'; //'+' - НЕ выводить при поиске ПУСТЫЕ организации, '-' - выводить все найденные
var printOffice = '-'; //'+' - записывать данные из поля Офис, '-' - не записывать
var printLastLogin = '-'; //'+' - записывать последнюю дату авторизации, '-' - не записывать
var skipFilials = '+'; //'+' - пропускать подразделения и их дочерние со словом "Филиал" в названии,'-' - не пропускать
var uniqueEmployees = '+'; //'+' - включить режим вывода в ЛОГ БЕЗ ВЫГРУЗКИ количества уникальных логинов, '-' - не включать

	// Оптимизированная функция поиска организаций
	function optimizedOrgSearch(getOrg, textData) {
		var fullName = getOrg.properties['lecm-orgstr:element-full-name'];
		var shortName = getOrg.properties['lecm-orgstr:element-short-name'];
		var isActive = getOrg.properties['lecm-dic:active'];
		if (isActive && 
			(getOrg.name.toLowerCase().includes(textData.toLowerCase()) || 
			 (shortName && shortName.toLowerCase().includes(textData.toLowerCase())))) {
			var children = getOrg.getChildren();
			var hasChildren = children && children.length > 0;
			var parent = getOrg.getParent();
			var output = [
				'\t' + fullName,
				'\t' + shortName,
				'\t' + getOrg.nodeRef
			].join('\n');
			if (hasChildren) {
				output += '\n\t🔰 В организации присутствуют сотрудники и/или подразделения';
			} else if (noEmpty == '-') {
				output += '\n\t❗ Организация пуста';
			} else {
				return; // Пропускаем пустые организации, если noEmpty == '+'
			}
			print(output);
			print('……………………………………………………………………………………………………………………………………………………………………………………………………………………');
		}
	}

	// Оптимизированная функция получения данных сотрудника
	function getOptimizedEmployeeData(employee, staffPos, org) {
		var personAssoc = employee.assocs['lecm-orgstr:employee-person-assoc'];
		var status;
		var lastLogin = null;
		if (personAssoc && personAssoc[0]) {
			var person = personAssoc[0];
			if (person.properties['cm:authorizationStatus'] + '' == 'AUTHORIZED') {
				status = 'Авторизована';
			} else {
				status = 'Никогда не была авторизована';
			}
			if (printLastLogin == '+') {
				lastLogin = person.properties['infbrt:lastSuccessfulLogin']?rnUtils.formatDate(person.properties['infbrt:lastSuccessfulLogin'], "dd.MM.yyyy"):'Нет даты';
			}
		} else {
			status = 'Нет связи с профилем';
		}
		if (isBoss != '+' || employee.properties['lecm-orgstr-rn-aspects:is-boss'] == true) {
			return compileEmployeeResult(org, staffPos, employee, status, lastLogin);
		}
		return '';
	}

	// Оптимизированная компиляция результата
	function compileEmployeeResult(org, staffPos, employee, status, lastLogin) {
		var loginData = employee.properties['cm:created'] ? 
			rnUtils.formatDate(employee.properties['cm:created'], "dd.MM.yyyy") : 
		'Нет даты';
		var result = [
			org,
			staffPos.properties['lecm-orgstr:element-member-position-assoc-text-content'],
			employee.properties['lecm-orgstr:employee-last-name'] + ' ' + 
			employee.properties['lecm-orgstr:employee-first-name'] + ' ' + 
			employee.properties['lecm-orgstr:employee-middle-name'],
			employee.properties['lecm-orgstr:employee-person-login'],
			employee.properties['lecm-orgstr:employee-email'],
			status,
			loginData,
			employee.nodeRef.toString()
		].join(';');
		if (printRoleProf == '+') {
			var empRoles = orgstructure.getEmployeeBusinessRoles(employee.nodeRef);
			result += ';' + getBusinessRoles(empRoles);
		}
		if (printOffice == '+') {
			result += ';' + employee.properties['lecm-orgstr-rn-aspects:office-assoc-text-content'];
		}
		if (printLastLogin == '+') {
			result += ';' + lastLogin;
		}
		return result + '\n';
	}

	// Оптимизированная функция подсчета сотрудников
	function optimizedEmployeeCounter(orgNode, uniqueLogins) {
		var children = orgNode.getChildren();
		if (!children) return;
		for (var i = 0; i < children.length; i++) {
			var child = children[i];
			var employeeLinks = child.getChildAssocsByType('lecm-orgstr:employee-link');
			if (employeeLinks && employeeLinks.length > 0) {
				var employeeElement = employeeLinks[0];
				if (employeeElement && employeeElement.assocs['lecm-orgstr:employee-link-employee-assoc']) {
					var pers = employeeElement.assocs['lecm-orgstr:employee-link-employee-assoc'][0];
					if (pers) {
						var email = pers.properties['lecm-orgstr:employee-email'] || '';
						var shortName = pers.properties['lecm-orgstr:employee-short-name'] || '';
						if (skipEngineers != '+' || 
							(email.toLowerCase().indexOf('example1') == -1 && 
							 email.toLowerCase().indexOf('example2') == -1 && 
							 email.toLowerCase().indexOf('example3') == -1 && 
							 shortName.toLowerCase().indexOf('info') == -1)) {
							var login = pers.properties["lecm-orgstr:employee-person-login"];
							if (login && uniqueLogins.indexOf(login) == -1) {
								uniqueLogins.push(login);
							}
						}
					}
				}
			}
			if (skipFilials == '+') {
				if (child.properties['lecm-orgstr:element-full-name'] && child.properties['lecm-orgstr:element-full-name'].indexOf('Филиал ') == -1) {
					optimizedEmployeeCounter(child, uniqueLogins);
				} else if (child.properties['lecm-orgstr:element-full-name'] && child.properties['lecm-orgstr:element-full-name'].indexOf('Филиал ') != -1) {
					print('Пропускаю "'+child.properties['lecm-orgstr:element-full-name']+'"... ('+child.nodeRef.toString()+')');
				}
			} else {
				optimizedEmployeeCounter(child, uniqueLogins);
			}
		}
	}

	function processOrganization(orgNode, fileReport) {
		if (skipFilials == '+') {
			if (orgNode.typeShort != 'lecm-orgstr:organization-unit' || orgNode.properties['lecm-orgstr:element-full-name'].indexOf('Филиал ') != -1) {
				if (orgNode.properties['lecm-orgstr:element-full-name'] && orgNode.properties['lecm-orgstr:element-full-name'].indexOf('Филиал ') != -1) {
					print('Пропускаю "'+orgNode.properties['lecm-orgstr:element-full-name']+'"... ('+orgNode.nodeRef.toString()+')');
				}
				return;
			}
		} else {
			if (orgNode.typeShort != 'lecm-orgstr:organization-unit') {
				return;
			}
		}
		var children = orgNode.getChildren();
		for (var i = 0; i < children.length; i++) {
			var child = children[i];
			if (child.typeShort == 'lecm-orgstr:staff-list') {
				processEmployee(child, orgNode.properties['lecm-orgstr:element-full-name'], fileReport);
			} else if (child.typeShort == 'lecm-orgstr:organization-unit') {
				processOrganization(child, fileReport);
			}
		}
	}

	function processEmployee(staffPos, orgName, fileReport) {
		var employeeLinks = staffPos.getChildAssocsByType('lecm-orgstr:employee-link');
		if (!employeeLinks || employeeLinks.length == 0) return;
		var employeeElement = employeeLinks[0];
		if (!employeeElement || !employeeElement.assocs['lecm-orgstr:employee-link-employee-assoc']) return;
		var pers = employeeElement.assocs['lecm-orgstr:employee-link-employee-assoc'][0];
		if (!pers) return;
		var email = pers.properties['lecm-orgstr:employee-email'] || '';
		var shortName = pers.properties['lecm-orgstr:employee-short-name'] || '';
		if (skipEngineers != '+' || 
			(email.toLowerCase().indexOf('example1') == -1 && 
			 email.toLowerCase().indexOf('example2') == -1 && 
			 email.toLowerCase().indexOf('example3') == -1 && 
			 shortName.toLowerCase().indexOf('info') == -1)) {
			var employeeData = getOptimizedEmployeeData(pers, staffPos, orgName);
			if (employeeData) {
				fileReport.content += employeeData;
			}
		}
	}

	function bytesConverter(file) {
		return file.size > 999 ? '[💾 '+(file.size/1024).toFixed(2)+' Kb]' : '[💾 '+file.size+' b]';
	}
	function getBusinessRoles(roles) {
		var profiles = [];
		for (var i = 0; i < roles.length; i++) {
			if (roles[i].name && roles[i].name.toLowerCase().includes('профиль')) {
				profiles.push(roles[i].name);
			}
		}
		return profiles.join(', ');
	}

	if (text == 'название_организации_или_нодрефа') {
		print('❗ Не заданы параметры поиска организации ❗');
	} else if (text.indexOf('ore/') == -1) {
		print('Найденые совпадения:\n—————————————————————————————————————————————————————————————————————————————————');
		var holdings = config.holdings || [config.holding];
		for (var i = 0; i < holdings.length; i++) {
			var holdingNode = search.findNode(holdings[i]);
			if (holdingNode) {
				var children = holdingNode.getChildren();
				for (var j = 0; j < children.length; j++) {
					optimizedOrgSearch(children[j], text);
				}
			}
		}
		print('❗ Если пусто, значит все организации по совпадению в названии с "'+text+'" - пустые');
		print('\nНеобходимо указать нодрефу нужной организации из полученного списка');
	} else if (text.substr(text.indexOf('ore/')+1, 36).length == 36) {
		var orgNode = search.findNode(text);
		if (!orgNode) {
			print('Организация не найдена');
		} else {
			var folderName = config.tempFolder || (config.folders && config.folders[0]) || '';
			var folder = space.childByNamePath(folderName);
			var addressPart1 = config.urlPrefix || (config.urls && config.urls[0]) || '';
			var tagItem = config.tag || (config.tags && config.tags[0]) || '';
			// Очистка старых файлов
			var files = folder.getChildren();
			for (var i = 0; i < files.length; i++) {
				var file = files[i];
				if (file.properties['cm:name'].indexOf('Штатное расписание') != -1) {
					print('Нашел существующий файл "' + file.name + '", удаляю...\n');
					file.remove();
				}
			}
			var orgName = (orgNode.properties['lecm-orgstr:element-short-name'] || 
						   orgNode.properties['lecm-orgstr-aspects:linked-organization-assoc-text-content'] || '')
			.replace(/"/g, '');
			if (uniqueEmployees != '+') {
				try {
					var fileName = "Штатное расписание " + orgName + " " + rnUtils.formatDate(new Date(), "dd.MM.yyyy") + '.csv';
					var fileReport = folder.createFile(fileName);
					fileReport.mimetype = "text/csv";
					var header = '\uFEFFПодразделение;Должность;ФИО;Логин;Почта;Статус УЗ;Дата создания;Нодрефа';
					if (printRoleProf == '+') {
						header += ';Профили';
					}
					if (printOffice == '+') {
						header += ';Офис';
					}
					if (printLastLogin == '+') {
						header += ';LastLogin';
					}
					fileReport.content = header + '\n';
					processOrganization(orgNode, fileReport);
					print('Создан новый файл, начинаю запись...\n\n📃 Имя файла: "'+fileReport.name+'" '+bytesConverter(fileReport)+'\n🌐 Адрес файла: ⇓\n' + addressPart1 + fileReport.nodeRef);
				} catch(error) {
					print('\n❌❌❌ '+error.message+' ❌❌❌');
				}
			} else {
				var uniqueLogins = [];
				optimizedEmployeeCounter(orgNode, uniqueLogins);
				print(orgName+' на '+rnUtils.formatDate(new Date(), "dd.MM.yyyy")+':');
				print("КСЭД " + tagItem + " -> " + uniqueLogins.length);
			}
		}
	} else {
		print('Ни одно из условий не подходит или нодрефа указана с ошибкой');
	}
